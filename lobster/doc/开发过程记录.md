### 任务 0.2.1-0.2.3：AI 执行链路打通

**时间**：2026-06-30

**修改文件**：
- `src/server/agent/aiExecutor.ts`
- `src/server/agent/agentRuntime.ts`

**主要改动**：

1. **aiExecutor.ts**：
   - 新增 `shouldUseRealAI()` 函数，优先检查 AI Gateway 是否有可用 Key
   - 修改 `USE_MOCK_BY_DEFAULT` 默认值，保留配置灵活性
   - 从 AI Gateway 获取 Key：`const keyConfig = aiGateway.getAvailableKey("openai")`
   - 使用 Gateway 的 Key 调用 OpenAI API
   - 调用后记录用量：`aiGateway.recordCall(keyConfig.id, tokensUsed)`
   - 新增 `keyId` 字段用于追踪

2. **agentRuntime.ts**：
   - 移除直接操作内部 Map 的 hack 代码
   - 改用标准的 `taskRepo.update()` 接口更新任务状态
   - 传入结构化数据：`{ status, result, completedAt }`

**代码示例**：

```typescript
// aiExecutor.ts - 从 Gateway 获取 Key
const keyConfig = aiGateway.getAvailableKey("openai");
if (!keyConfig) {
  return { success: false, content: "AI Gateway 没有可用 Key", ... };
}

// 调用后记录用量
aiGateway.recordCall(keyConfig.id, tokensUsed);

// agentRuntime.ts - 通过 Repository 更新
await taskRepo.update(taskId, {
  status: result.success ? "DONE" : "FAILED",
  result: result.content?.slice(0, 1000) ?? null,
  completedAt: new Date(),
});
```

**验收结果**：
- ✅ AI Gateway 集成完成
- ✅ Mock 模式可通过配置切换
- ✅ 执行结果通过 Repository 接口写回

```markdown
# 任务：实现 Vault Service

## 目标
提供凭证的 CRUD 操作，自动加密存储，保护明文不可见。

## 实现步骤

### 1. 创建 Vault Service
文件：`src/server/vault/vaultService.ts`

```typescript
import { prisma } from "~/server/db";
import crypto from "crypto";

const ENCRYPTION_KEY = process.env.VAULT_ENCRYPTION_KEY || crypto.randomBytes(32).toString("hex");
const ALGORITHM = "aes-256-gcm";

export class VaultService {
  private encrypt(plaintext: string): string {
    const iv = crypto.randomBytes(16);
    const cipher = crypto.createCipheriv(ALGORITHM, Buffer.from(ENCRYPTION_KEY.slice(0, 32)), iv);
    let encrypted = cipher.update(plaintext, "utf8", "hex");
    encrypted += cipher.final("hex");
    const authTag = cipher.getAuthTag();
    return `${iv.toString("hex")}:${authTag.toString("hex")}:${encrypted}`;
  }

  private decrypt(ciphertext: string): string {
    const [ivHex, authTagHex, encrypted] = ciphertext.split(":");
    const iv = Buffer.from(ivHex!, "hex");
    const authTag = Buffer.from(authTagHex!, "hex");
    const decipher = crypto.createDecipheriv(ALGORITHM, Buffer.from(ENCRYPTION_KEY.slice(0, 32)), iv);
    decipher.setAuthTag(authTag);
    let decrypted = decipher.update(encrypted!, "hex", "utf8");
    decrypted += decipher.final("utf8");
    return decrypted;
  }

  async createCredential(input: {
    tenantId: string;
    name: string;
    type: string;
    systemType: string;
    credentialData: string;
    ownerId?: string;
    agentIds?: string[];
    expiresAt?: Date;
  }) {
    const encryptedData = this.encrypt(input.credentialData);

    return prisma.credential.create({
      data: {
        tenantId: input.tenantId,
        name: input.name,
        type: input.type,
        systemType: input.systemType,
        encryptedData,
        ownerId: input.ownerId,
        agentIds: JSON.stringify(input.agentIds ?? []),
        expiresAt: input.expiresAt,
      },
    });
  }

  async listCredentials(tenantId: string) {
    const creds = await prisma.credential.findMany({
      where: { tenantId, isActive: true },
      orderBy: { createdAt: "desc" },
    });

    return creds.map((c) => ({
      id: c.id,
      tenantId: c.tenantId,
      name: c.name,
      type: c.type,
      systemType: c.systemType,
      ownerId: c.ownerId,
      agentIds: JSON.parse(c.agentIds),
      isActive: c.isActive,
      expiresAt: c.expiresAt?.toISOString() ?? null,
      lastUsedAt: c.lastUsedAt?.toISOString() ?? null,
      createdAt: c.createdAt.toISOString(),
      updatedAt: c.updatedAt.toISOString(),
    }));
  }

  async getCredentialData(credentialId: string, requesterAgentId: string): Promise<string | null> {
    const cred = await prisma.credential.findUnique({
      where: { id: credentialId },
    });

    if (!cred || !cred.isActive) return null;

    const agentIds = JSON.parse(cred.agentIds) as string[];
    if (!agentIds.includes(requesterAgentId)) {
      return null;
    }

    if (cred.expiresAt && cred.expiresAt < new Date()) {
      return null;
    }

    await prisma.credential.update({
      where: { id: credentialId },
      data: { lastUsedAt: new Date() },
    });

    return this.decrypt(cred.encryptedData);
  }

  async deleteCredential(credentialId: string) {
    return prisma.credential.update({
      where: { id: credentialId },
      data: { isActive: false },
    });
  }
}

export const vaultService = new VaultService();

import type { ToolDefinition, ToolCallResult, ToolCallContext, IToolRegistry, IToolExecutor } from "./types";
import { BaseToolExecutor } from "./registry";

export class InMemoryToolRegistry implements IToolRegistry {
  private tools: Map<string, ToolDefinition> = new Map();
  private executors: Map<string, IToolExecutor> = new Map();

  register(definition: ToolDefinition, executor: IToolExecutor): void {
    this.tools.set(definition.id, definition);
    this.executors.set(definition.id, executor);
  }

  getAllTools(): ToolDefinition[] {
    return Array.from(this.tools.values());
  }

  getToolsBySystem(systemType: string): ToolDefinition[] {
    return Array.from(this.tools.values()).filter((t) => t.systemType === systemType);
  }

  getTool(id: string): ToolDefinition | undefined {
    return this.tools.get(id);
  }

  search(query: string, limit = 10): ToolDefinition[] {
    const q = query.toLowerCase();
    return Array.from(this.tools.values())
      .filter(
        (t) =>
          t.name.toLowerCase().includes(q) ||
          t.description.toLowerCase().includes(q)
      )
      .slice(0, limit);
  }

  async execute(toolId: string, params: Record<string, unknown>, context: ToolCallContext): Promise<ToolCallResult> {
    const tool = this.tools.get(toolId);
    const executor = this.executors.get(toolId);

    if (!tool || !executor) {
      return { success: false, error: `Tool ${toolId} 不存在`, executionTimeMs: 0 };
    }

    if (!executor.validate(params, tool)) {
      return { success: false, error: "参数验证失败", executionTimeMs: 0 };
    }

    const startTime = Date.now();
    try {
      const result = await executor.execute(tool, params, context);
      return { ...result, executionTimeMs: Date.now() - startTime };
    } catch (error) {
      return {
        success: false,
        error: error instanceof Error ? error.message : "执行失败",
        executionTimeMs: Date.now() - startTime,
      };
    }
  }
}

function sanitizeParams(params: Record<string, unknown>): Record<string, unknown> {
  const sensitiveParams = ["password", "token", "secret", "key", "apiKey", "api_key"];
  const sanitized = { ...params };

  for (const key of Object.keys(sanitized)) {
    if (sensitiveParams.some((s) => key.toLowerCase().includes(s))) {
      sanitized[key] = "***REDACTED***";
    }
  }

  return sanitized;
}

import { BaseToolExecutor } from "~/server/tools/registry";
import type { ToolDefinition, ToolCallResult, ToolCallContext } from "~/server/tools/types";
import { vaultService } from "~/server/vault/vaultService";

interface HttpConnectorConfig {
  baseUrl: string;
  credentialId?: string;
  defaultHeaders?: Record<string, string>;
  timeout?: number;
}

export class HttpConnector extends BaseToolExecutor {
  private config: HttpConnectorConfig;

  constructor(config: HttpConnectorConfig) {
    super();
    this.config = { timeout: 30000, ...config };
  }

  async execute(
    definition: ToolDefinition,
    params: unknown,
    context: ToolCallContext
  ): Promise<ToolCallResult> {
    const p = params as { method: string; path: string; headers?: Record<string, string>; body?: unknown; query?: Record<string, string> };
    const startTime = Date.now();

    try {
      let authHeaders: Record<string, string> = {};
      if (this.config.credentialId) {
        const token = await vaultService.getCredentialData(this.config.credentialId, context.agentId);
        if (token) {
          authHeaders = { Authorization: `Bearer ${token}` };
        }
      }

      const url = new URL(p.path, this.config.baseUrl);
      if (p.query) {
        Object.entries(p.query).forEach(([k, v]) => url.searchParams.append(k, v));
      }

      const headers = {
        "Content-Type": "application/json",
        ...this.config.defaultHeaders,
        ...authHeaders,
        ...p.headers,
      };

      const controller = new AbortController();
      const timeoutId = setTimeout(() => controller.abort(), this.config.timeout);

      const response = await fetch(url.toString(), {
        method: p.method,
        headers,
        body: p.body ? JSON.stringify(p.body) : undefined,
        signal: controller.signal,
      });

      clearTimeout(timeoutId);

      let data: unknown;
      const contentType = response.headers.get("content-type");
      if (contentType?.includes("application/json")) {
        data = await response.json();
      } else {
        data = await response.text();
      }

      if (!response.ok) {
        return { success: false, error: `HTTP ${response.status}: ${response.statusText}`, data, executionTimeMs: Date.now() - startTime };
      }

      return { success: true, data, executionTimeMs: Date.now() - startTime };
    } catch (error) {
      if (error instanceof Error && error.name === "AbortError") {
        return { success: false, error: "请求超时", executionTimeMs: Date.now() - startTime };
      }
      return { success: false, error: error instanceof Error ? error.message : "请求失败", executionTimeMs: Date.now() - startTime };
    }
  }
}

        subTask: task,
        approver: decision.approver,
        urgency: decision.urgency,
      });

      // 等待审批结果（轮询或 WebSocket）
      const result = await this.waitForApproval(approval.id, {
        timeout: decision.timeout ?? 3600  // 默认 1 小时
      });

      if (result.decision === 'rejected') {
        throw new ApprovalRejectedError(task, result.reason);
      }
      // approved 继续执行
    }

    // 执行子任务
    await this.executeSubTask(task, context);
  }
}

```typescript
/**
 * 监听接管任务完成，继续原任务执行
 */
async waitForIntervention(taskId: string, timeout: number): Promise<void> {
  const startTime = Date.now();

  while (Date.now() - startTime < timeout) {
    const task = await taskService.getTask(taskId);

    if (task.status === 'completed') {
      // 恢复原任务
      const result = task.result;
      if (result.action === 'resume') {
        await this.resumeTask(result.originalTaskId, result.resumeFrom);
      } else if (result.action === 'abort') {
        throw new TaskAbortedError('人工决定终止任务');
      } else if (result.action === 'skip') {
        // 跳过失败的子任务继续执行
        await this.skipAndContinue(result.originalTaskId, result.skipSubTaskId);
      }
      return;
    }

    if (task.status === 'expired' || task.status === 'rejected') {
      throw new InterventionTimeoutError('人工接管超时');
    }

    await this.sleep(5000);  // 每 5 秒检查一次
  }

  throw new InterventionTimeoutError('等待人工接管超时');
}
